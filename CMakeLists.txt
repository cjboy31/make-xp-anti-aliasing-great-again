cmake_minimum_required(VERSION 3.20)
project(xplane_native_aa VERSION 0.1.0 LANGUAGES CXX)

add_library(native_aa_core STATIC src/frame_contract.cpp)
target_include_directories(native_aa_core PUBLIC include)
target_compile_features(native_aa_core PUBLIC cxx_std_17)
set_target_properties(native_aa_core PROPERTIES CXX_EXTENSIONS OFF)

function(aa_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endfunction()
aa_warnings(native_aa_core)

include(CTest)
if(BUILD_TESTING)
  add_executable(frame_contract_tests tests/frame_contract_tests.cpp)
  target_link_libraries(frame_contract_tests PRIVATE native_aa_core)
  aa_warnings(frame_contract_tests)
  add_test(NAME native-aa-frame-contract COMMAND frame_contract_tests)
endif()
