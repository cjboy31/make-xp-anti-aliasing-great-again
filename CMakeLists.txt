cmake_minimum_required(VERSION 3.20)
project(xplane_native_aa VERSION 0.1.0 LANGUAGES CXX)

add_library(native_aa_core STATIC src/frame_contract.cpp src/motion.cpp)
target_include_directories(native_aa_core PUBLIC include)
target_compile_features(native_aa_core PUBLIC cxx_std_17)
set_target_properties(native_aa_core PROPERTIES CXX_EXTENSIONS OFF)

function(aa_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endfunction()
aa_warnings(native_aa_core)

include(CTest)
if(BUILD_TESTING)
  add_executable(frame_contract_tests tests/frame_contract_tests.cpp)
  target_link_libraries(frame_contract_tests PRIVATE native_aa_core)
  aa_warnings(frame_contract_tests)
  add_test(NAME native-aa-frame-contract COMMAND frame_contract_tests)
  add_executable(motion_tests tests/motion_tests.cpp)
  target_link_libraries(motion_tests PRIVATE native_aa_core)
  aa_warnings(motion_tests)
  add_test(NAME native-aa-motion-contract COMMAND motion_tests)
endif()

option(NATIVE_AA_BUILD_VULKAN_TESTS "Build the offscreen SDR Vulkan prototype" OFF)
option(NATIVE_AA_GPU_VALIDATION "Require synchronization validation for GPU tests" ON)
if(NATIVE_AA_BUILD_VULKAN_TESTS)
  find_package(Vulkan 1.1 REQUIRED)
  find_program(GLSLANG_VALIDATOR NAMES glslangValidator glslang REQUIRED)
  add_library(native_aa_vulkan STATIC src/vulkan/smaa_pipeline.cpp src/vulkan/camera_motion_pipeline.cpp)
  target_include_directories(native_aa_vulkan PUBLIC include)
  target_compile_features(native_aa_vulkan PUBLIC cxx_std_17)
  target_link_libraries(native_aa_vulkan PUBLIC Vulkan::Vulkan native_aa_core)
  aa_warnings(native_aa_vulkan)
  if(NOT MSVC)
    target_compile_options(native_aa_vulkan PRIVATE -Wno-missing-field-initializers)
  endif()
  set(aa_spirv)
  foreach(pass RANGE 0 2)
    foreach(stage vert frag)
      set(output "${CMAKE_CURRENT_BINARY_DIR}/shaders/smaa-${pass}.${stage}.spv")
      add_custom_command(OUTPUT "${output}"
        COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/shaders"
        COMMAND "${GLSLANG_VALIDATOR}" -V --target-env vulkan1.1 "-DAA_PASS=${pass}"
          "-I${CMAKE_CURRENT_SOURCE_DIR}/third_party/smaa"
          "${CMAKE_CURRENT_SOURCE_DIR}/shaders/smaa.${stage}" -o "${output}"
        DEPENDS "shaders/smaa.${stage}" third_party/smaa/SMAA.hlsl VERBATIM)
      list(APPEND aa_spirv "${output}")
    endforeach()
  endforeach()
  set(motion_spirv "${CMAKE_CURRENT_BINARY_DIR}/shaders/camera-motion.comp.spv")
  add_custom_command(OUTPUT "${motion_spirv}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/shaders"
    COMMAND "${GLSLANG_VALIDATOR}" -V --target-env vulkan1.1
      "${CMAKE_CURRENT_SOURCE_DIR}/shaders/camera_motion.comp" -o "${motion_spirv}"
    DEPENDS shaders/camera_motion.comp VERBATIM)
  add_custom_target(smaa_shaders DEPENDS ${aa_spirv} "${motion_spirv}")
  add_dependencies(native_aa_vulkan smaa_shaders)
  if(BUILD_TESTING)
    add_executable(smaa_vulkan_tests tests/smaa_vulkan_tests.cpp)
    target_include_directories(smaa_vulkan_tests PRIVATE third_party/smaa)
    target_link_libraries(smaa_vulkan_tests PRIVATE native_aa_vulkan)
    aa_warnings(smaa_vulkan_tests)
    add_dependencies(smaa_vulkan_tests smaa_shaders)
    if(NOT MSVC)
      target_compile_options(smaa_vulkan_tests PRIVATE -Wno-missing-field-initializers)
    endif()
    set(validation)
    if(NATIVE_AA_GPU_VALIDATION)
      set(validation --validation)
    endif()
    add_test(NAME native-aa-vulkan-synthetic COMMAND smaa_vulkan_tests
      --shaders "${CMAKE_CURRENT_BINARY_DIR}/shaders"
      --output "${CMAKE_CURRENT_BINARY_DIR}/gpu-evidence" ${validation})
    set_tests_properties(native-aa-vulkan-synthetic PROPERTIES TIMEOUT 120)
  endif()
endif()
